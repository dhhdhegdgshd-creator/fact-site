function showFacts() {

    const container = document.getElementById("facts");
    const search = document.getElementById("search").value.toLowerCase();
    const category = document.getElementById("category").value;

    container.innerHTML = "";

    facts.forEach((fact, index) => {

        const matchesSearch =
            fact.text.toLowerCase().includes(search);

        const matchesCategory =
            category === "all" ||
            fact.category === category;

        if (matchesSearch && matchesCategory) {

            container.innerHTML += `
                <div class="card">
                    <span class="category">${fact.category}</span>
                    <h3>💡 Интересный факт</h3>
                    <p>${fact.text}</p>
                    <button class="like"
                        onclick="like(this)">
                        ❤️ Мне нравится
                    </button>
                </div>
            `;
        }
    });
}

function randomFact() {

    const random =
        facts[Math.floor(Math.random() * facts.length)];

    document.getElementById("dailyFact").textContent =
        random.text;
}

function like(button) {

    if (button.textContent.includes("❤️")) {
        button.textContent = "💖 Нравится!";
    } else {
        button.textContent = "❤️ Мне нравится";
    }
}

function toggleTheme() {

    document.body.classList.toggle("light");

    if (document.body.classList.contains("light")) {
        localStorage.setItem("theme", "light");
    } else {
        localStorage.setItem("theme", "dark");
    }
}

if (localStorage.getItem("theme") === "light") {
    document.body.classList.add("light");
}
